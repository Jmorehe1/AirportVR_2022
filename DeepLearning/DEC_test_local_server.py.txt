# Define a function to tokenize words for filtering responses
def get_tokens_as_list(word_list):
    tokens_list = [tokenizer([word], add_special_tokens=False).input_ids[0] for word in word_list]
    return tokens_list

# Banned words for response generation
bad_words_ids = get_tokens_as_list(["¡", "Great", "Perfecto", "Entendido", "!"])

# Function to handle response generation in Spanish
def forward_pass(messages):
    while True:
        inputs = tokenizer(
            messages, return_tensors="pt", padding=True, truncation=True
        ).to(device)

            bad_words_ids=bad_words_ids,
            do_sample=True,

            max_new_tokens=512
        )
